CREATE OR ALTER PROCEDURE SPR_EBS_Application
AS
BEGIN
	SET XACT_ABORT, NOCOUNT ON;
	SET DATEFORMAT ymd

	DECLARE @AcademicYear VARCHAR(5)

	SET @AcademicYear = (
		SELECT
			CFG.Value
		FROM Config CFG
		WHERE
			CFG.ConfigID = 'EBS_AppAcademicYearID'
	)

	--Get Current Year Dates
	DROP TABLE IF EXISTS #CurrentYear
	SELECT
		AcademicYear = @AcademicYear,
		FundingYear = FY.FUNDING_YEAR,
		YrStart = FY.START_DATE,
		YrEnd = FY.END_DATE
	INTO #CurrentYear
	FROM ebs.dbo.FUNDING_YEARS FY
	WHERE
		CAST ( FY.START_DATE AS DATE ) <= '20' + RIGHT ( @AcademicYear, 2 ) + '-07-31'
		AND CAST ( FY.END_DATE AS DATE ) >= '20' + LEFT ( @AcademicYear, 2 ) + '-08-01'


	--Main Addresses
	DROP TABLE IF EXISTS #Addresses
	SELECT
		PersonCode = P.PERSON_CODE,
		AddressCode = AD.ADDRESS_CODE,
		RowNum = 
			ROW_NUMBER () OVER (
				PARTITION BY
					P.PERSON_CODE
				ORDER BY
					CASE
						WHEN AD.END_DATE IS NULL THEN 1
						ELSE 2
					END,
					CASE
						WHEN AD.ADDRESS_TYPE = 'H' THEN 1
						WHEN AD.ADDRESS_TYPE = 'T' THEN 2
						WHEN AD.ADDRESS_TYPE = 'A' THEN 3
						WHEN AD.ADDRESS_TYPE = 'P' THEN 4
						WHEN AD.ADDRESS_TYPE = 'N' THEN 5
						WHEN AD.ADDRESS_TYPE = 'O' THEN 6
						ELSE 9
					END
			)
		INTO #Addresses
	FROM ebs.dbo.PEOPLE P
	INNER JOIN ebs.dbo.ADDRESSES AD
		ON AD.PERSON_CODE = P.PERSON_CODE
		AND AD.OWNER_TYPE = 'P'


	--Contacts
	DROP TABLE IF EXISTS #Contacts
	SELECT
		CON.PersonCode,
		TrainingConsultant1StaffRef = MAX ( CASE WHEN CON.Relationship = 'TC' THEN CON.StaffRef ELSE NULL END ),
		TrainingConsultant1StaffName = MAX ( CASE WHEN CON.Relationship = 'TC' THEN CON.StaffName ELSE NULL END ),
		TrainingConsultant2StaffRef = MAX ( CASE WHEN CON.Relationship = 'STC' THEN CON.StaffRef ELSE NULL END ),
		TrainingConsultant2StaffName = MAX ( CASE WHEN CON.Relationship = 'STC' THEN CON.StaffName ELSE NULL END ),
		BusinessDevelopmentStaffRef = MAX ( CASE WHEN CON.Relationship = 'BD' THEN CON.StaffRef ELSE NULL END ),
		BusinessDevelopmentStaffName = MAX ( CASE WHEN CON.Relationship = 'BD' THEN CON.StaffName ELSE NULL END )
		INTO #Contacts
	FROM (
		SELECT
			PersonCode = P.PERSON_CODE,
			Relationship = PL.RELATIONSHIP,
			StaffRef = STF.PERSON_CODE,
			StaffName = STF.FORENAME + ' ' + STF.SURNAME + ' (' + CAST ( STF.PERSON_CODE AS VARCHAR(50) ) + ')',
			RowNum =
				ROW_NUMBER () OVER (
					PARTITION BY
						P.PERSON_CODE,
						PL.RELATIONSHIP
					ORDER BY
						PL.START_DATE DESC,
						CASE
							WHEN PL.END_DATE IS NULL THEN 1 
							ELSE 2 
						END,
						P.PERSON_CODE DESC
				)
		FROM ebs.dbo.PEOPLE P
		INNER JOIN ebs.dbo.PEOPLE_PEOPLE_LINKS PL
			ON PL.PERSON_CODE_TO = P.PERSON_CODE
			AND PL.RELATIONSHIP IN (
				'TC', --Training Consultant (in charge of delivery aspect)
				'STC', --Secondary Training Consultant
				'BD' --Business Development (manages changes to employer)
			)
		INNER JOIN ebs.dbo.PEOPLE STF
			ON STF.PERSON_CODE = PL.PERSON_CODE_FROM
		WHERE
			CAST ( COALESCE ( PL.END_DATE, GETDATE() ) AS DATE ) >= CAST ( GETDATE() AS DATE )
	) CON
	WHERE
		CON.RowNum = 1
	GROUP BY
		CON.PersonCode


	--Main Disability
	DROP TABLE IF EXISTS #Disabilities
	SELECT
		DisabilityID = DIS.ID,
		PersonCode = DIS.PER_PERSON_CODE,
		DisabilityTypeCode = DIS.DISABILITY_TYPE,
		DisabilityTypeName = DISD.FES_LONG_DESCRIPTION,
		StartDate = DIS.START_DATE,
		EndDate = DIS.END_DATE,
		IsPrimaryNeed = DIS.PRIMARY_NEED,
		SpecialExamNeeds = DIS.SPECIAL_EXAM_NEEDS,
		IsAssistanceRequired = DIS.IS_ASSISTANCE_REQUIRED,
		IncludeInILR = DIS.DATA_RETURN,
		IsCurrent = 
			CASE
				WHEN 
					CAST ( COALESCE ( DIS.START_DATE, GETDATE() ) AS DATE ) <= CAST ( GETDATE() AS DATE )
					AND CAST ( COALESCE ( DIS.END_DATE, GETDATE() ) AS DATE ) >= CAST ( GETDATE() AS DATE )
					THEN 'Y'
				ELSE 'N'
			END,
		CreatedBy = DIS.CREATED_BY,
		CreatedDate = DIS.CREATED_DATE,
		RowNum = 
			ROW_NUMBER () OVER (
				PARTITION BY
					DIS.PER_PERSON_CODE
				ORDER BY
					DIS.PRIMARY_NEED DESC,
					DIS.DATA_RETURN DESC,
					CASE
						WHEN 
							CAST ( COALESCE ( DIS.START_DATE, GETDATE() ) AS DATE ) <= CAST ( GETDATE() AS DATE )
									AND CAST ( COALESCE ( DIS.END_DATE, GETDATE() ) AS DATE ) >= CAST ( GETDATE() AS DATE )
									THEN 'Y'
						ELSE 'N'
					END DESC,
					DIS.SPECIAL_EXAM_NEEDS DESC,
					DIS.IS_ASSISTANCE_REQUIRED DESC,
					DIS.CREATED_DATE DESC,
					DIS.ID DESC
			)
		INTO #Disabilities
	FROM ebs.dbo.DISABILITIES DIS
	INNER JOIN ebs.dbo.VERIFIERS DISD
		ON DISD.LOW_VALUE = DIS.DISABILITY_TYPE
		AND DISD.RV_DOMAIN = 'DISABILITY_TYPE'
	WHERE
		CAST ( COALESCE ( DIS.START_DATE, '20' + LEFT ( @AcademicYear, 2 ) + '-08-01' ) AS DATE ) <= CAST ( '20' + LEFT ( @AcademicYear, 2 ) + '-07-31' AS DATE )
		AND CAST ( COALESCE ( DIS.END_DATE, '20' + RIGHT ( @AcademicYear, 2 ) + '-07-31' ) AS DATE ) >= CAST ( '20' + LEFT ( @AcademicYear, 2 ) + '-08-01' AS DATE )
		AND ISNUMERIC ( DIS.DISABILITY_TYPE ) = 1 --Exclude non-ILR codes


	--Main Query
	DROP TABLE IF EXISTS #Apps
	SELECT
		AcademicYearID = @AcademicYear,
		YearStart = CY.YrStart,
		YearEnd = CY.YrEnd,
		LearnRefNumber = CAST ( P.PERSON_CODE AS VARCHAR(50) ),
		LearnRefNumberILR = COALESCE ( P.FES_SEC_PERSON_CODE, CAST ( P.PERSON_CODE AS VARCHAR(50) ) ),
		Surname = P.SURNAME,
		Forename = P.FORENAME,
		DOB = P.DATE_OF_BIRTH,
		Age31Aug = 
			CASE
				WHEN MONTH ( P.DATE_OF_BIRTH ) <= 8 THEN '20' + LEFT ( @AcademicYear, 2 ) - YEAR ( P.DATE_OF_BIRTH )
				ELSE CAST ('20' + LEFT ( @AcademicYear, 2 ) AS INT ) - 1 - YEAR ( P.DATE_OF_BIRTH )
			END,
		AgeGroup = 
			CASE
				WHEN 
					CASE
						WHEN MONTH ( P.DATE_OF_BIRTH ) <= 8 THEN '20' + LEFT ( @AcademicYear, 2 ) - YEAR ( P.DATE_OF_BIRTH )
						ELSE CAST ('20' + LEFT ( @AcademicYear, 2 ) AS INT ) - 1 - YEAR ( P.DATE_OF_BIRTH )
					END
					< 16 THEN '14-15'
				WHEN 
					CASE
						WHEN MONTH ( P.DATE_OF_BIRTH ) <= 8 THEN '20' + LEFT ( @AcademicYear, 2 ) - YEAR ( P.DATE_OF_BIRTH )
						ELSE CAST ('20' + LEFT ( @AcademicYear, 2 ) AS INT ) - 1 - YEAR ( P.DATE_OF_BIRTH )
					END
					BETWEEN 16 AND 18 THEN '16-18'
				WHEN 
					CASE
						WHEN MONTH ( P.DATE_OF_BIRTH ) <= 8 THEN '20' + LEFT ( @AcademicYear, 2 ) - YEAR ( P.DATE_OF_BIRTH )
						ELSE CAST ('20' + LEFT ( @AcademicYear, 2 ) AS INT ) - 1 - YEAR ( P.DATE_OF_BIRTH )
					END
					BETWEEN 19 AND 23 THEN '19-23'
				WHEN 
					CASE
						WHEN MONTH ( P.DATE_OF_BIRTH ) <= 8 THEN '20' + LEFT ( @AcademicYear, 2 ) - YEAR ( P.DATE_OF_BIRTH )
						ELSE CAST ('20' + LEFT ( @AcademicYear, 2 ) AS INT ) - 1 - YEAR ( P.DATE_OF_BIRTH )
					END 
					>= 24 THEN '24+'
			END,
		ULN = P.UNIQUE_LEARN_NO,
		Address1 = AD.ADDRESS_LINE_1,
		Address2 = AD.ADDRESS_LINE_2,
		Address3 = AD.ADDRESS_LINE_3,
		Address4 = AD.ADDRESS_LINE_4,
		Address5 = AD.ADDRESS_LINE_5,
		Town = AD.TOWN,
		County = AD.REGION,
		PostCode = AD.UK_POST_CODE_PT1 + ' ' + AD.UK_POST_CODE_PT2,
		TelMobile = P.MOBILE_PHONE_NUMBER,
		TelHome = AD.TELEPHONE,
		EmailCollege = P.COLLEGE_EMAIL,
		EmailPersonal = P.PERSONAL_EMAIL,
		EmailOther = AD.EMAIL_ADDRESS,
		EmergencyContact1Name = P.FES_NEXT_OF_KIN,
		EmergencyContact1Number = P.MOBILE_PHONE_NUMBER,
		EmergencyContact2Name = P.FES_USER_47,
		EmergencyContact2Number = P.FES_USER_48,
		SexCode = P.SEX,
		SexName = SEX.FES_LONG_DESCRIPTION,
		DeclaredSex = PAD.DECLARED_SEX,
		GenderCode = PAD.GENDER,
		GenderName = GEN.FES_LONG_DESCRIPTION,
		Pronouns = PAD.PRONOUNS,
		JobTitle = PAD.JOB_TITLE,
		EthnicityCode = COALESCE ( P.ETHNICITY, 99 ),
		EthnicityName = ETH.DESCRIPTION,
		EthnicGroup = 
			CASE
				WHEN COALESCE ( P.ETHNICITY, 99 ) = 99 THEN 'Not Provided'
				WHEN COALESCE ( P.ETHNICITY, 99 ) = 31 THEN 'White British'
				WHEN COALESCE ( P.ETHNICITY, 99 ) = 34 THEN 'White'
				ELSE 'Ethnic Minority'
			END,
		EthnicGroupSort = 
			CASE
				WHEN COALESCE ( P.ETHNICITY, 99 ) = 99 THEN 1
				WHEN COALESCE ( P.ETHNICITY, 99 ) = 31 THEN 4
				WHEN COALESCE ( P.ETHNICITY, 99 ) = 34 THEN 3
				ELSE 2
			END,
		FreeSchoolMealsCode = 
			CASE
				WHEN PI.FREE_MEAL_ELIGIBILITY IN ( 'Y', '1', '2' ) THEN 'Y'
				ELSE 'N'
			END,
		FreeSchoolMealsName = 
			CASE
				WHEN PI.FREE_MEAL_ELIGIBILITY IN ( 'Y', '1', '2' ) THEN 'Eligible for Free School Meals'
				ELSE 'Not Eligible for Free School Meals'
			END,
		InCare = PAD.IS_IN_CARE,
		IsCarer = PAD.IS_CARER,
		HasCriminalConvictions = PAD.HAS_CRIMINAL_CONVICTIONS,
		InterviewAssistanceRequired = PAD.IS_INTERVIEW_ASSIST_REQUIRED,
		EHCPCode = PI.HAS_HEALTHCARE_PLAN,
		EHCPName = 
			CASE
				WHEN PI.HAS_HEALTHCARE_PLAN = 'Y' THEN 'Has Educational Health Care Plan'
				ELSE 'Does Not Have Educational Health Care Plan'
			END,
		HighNeedsCode = PI.HIGH_NEEDS,
		HighNeedsName = 
			CASE
				WHEN PI.HIGH_NEEDS = 'Y' THEN 'Is a High Needs Learner'
				ELSE 'Not a High Needs Learner'
			END,
		LearnDifDisCode = COALESCE ( P.LEARN_DIFF, 9 ),
		LearnDifDisName = DIF.DESCRIPTION,
		LearnDifAssess = 
			CASE
				WHEN PI.LEARN_DIFF_ASSMT = 'Y' THEN 'Y'
				ELSE 'N'
			END,
		MainDifficultyDisabilityCode = DIS.DisabilityTypeCode,
		MainDifficultyDisabilityName = DIS.DisabilityTypeName,
		GCSEMathsGrade = PI.GCSE_MATHS_QUAL_GRADE,
		GCSEEnglishGrade = PI.GCSE_ENGLISH_QUAL_GRADE,
		FacultyCode = FAC.ORGANISATION_CODE,
		FacultyName = FAC.FES_FULL_NAME,
		FacultyHeadStaffCode = FACO.FES_STAFF_CODE,
		FacultyHeadSurname = FACO.SURNAME,
		FacultyHeadForename = FACO.FORENAME,
		FacultyHeadName = FACO.FORENAME + ' ' + FACO.SURNAME + ' (' + FACO.FES_STAFF_CODE + ')',
		TeamCode = UIO.OFFERING_ORGANISATION,
		TeamName = TEAM.FES_FULL_NAME,
		TeamHeadStaffCode = TEAMO.FES_STAFF_CODE,
		TeamHeadSurname = TEAMO.SURNAME,
		TeamHeadForename = TEAMO.FORENAME,
		TeamHeadName = TEAMO.FORENAME + ' ' + TEAMO.SURNAME + ' (' + TEAMO.FES_STAFF_CODE + ')',
		StudyLocationCode = UIO.SLOC_LOCATION_CODE,
		StudyLocationName = LOC.FES_LONG_DESCRIPTION,
		SSA1Code = FQA.SSA_TIER1_CODE,
		SSA1Name = RIGHT ( '00' + FQA.SSA_TIER1_CODE, 2 ) + ' - ' + FQA.SSA_TIER1_DESCRIPTION,
		SSA2Code = FQA.SSA_TIER2_CODE,
		SSA2Name = RIGHT ( '00' + FQA.SSA_TIER2_CODE, 4 ) + ' - ' + FQA.SSA_TIER2_DESCRIPTION,
		ApplicationID = PU.ID,
		ParentID = UI.UI_ID,
		ParentTitle = UI.FULL_DESCRIPTION,
		CourseID = UIO.UIO_ID,
		CourseCode = UIO.FES_UINS_INSTANCE_CODE,
		CourseTitle = UIO.LONG_DESCRIPTION,
		OccurrenceCode = UIO.CALOCC_OCCURRENCE_CODE,
		GroupCode = COALESCE ( TUT.Name, 'None' ),
		LearnAimRef = UQA.QUAL_CODE,
		LearnAimTitle = FQA.DESCRIPTION,
		IsSubcontracted = 
			CASE
				WHEN FAC.ORGANISATION_CODE = 'PART' THEN 'Y'
				ELSE 'N'
			END,
		SubcontractorName = 
			CASE
				WHEN FAC.ORGANISATION_CODE = 'PART' THEN REPLACE ( TEAM.FES_FULL_NAME, 'Partnerships - ', '' )
				ELSE NULL
			END,
		AwardBody = FQA.AWARD_BODY_1,
		LearningAimTypeCode = FQA.QUAL_TYPE,
		LearningAimTypeName = FQA.QUAL_TYPE_DESCRIPTION,
		QualTypeCode = UI.FES_QUALIFICATION_TYPE,
		QualTypeName = QT.FES_LONG_DESCRIPTION,
		SourceOfFinanceCode = UI.FES_SOURCE_FINANCE,
		SourceOfFinanceName = FT.FES_LONG_DESCRIPTION,
		ModeOfAttendanceCode = UIO.FES_MOA_CODE,
		ModeOfAttendanceName = MOA.FES_LONG_DESCRIPTION,
		CompletionStatusCode = 
			COALESCE (
				CMP.LSC_VALUE,
				9
			),
		CompletionStatusName = 
			CASE
				COALESCE (
					CMP.LSC_VALUE,
					9
				)
				WHEN 1 THEN 'Continuing'
				WHEN 2 THEN 'Completed'
				WHEN 3 THEN 'Withdrawn'
				WHEN 4 THEN 'Transferred'
				WHEN 6 THEN 'Break in Learning'
				ELSE NULL
			END,
		OutcomeCode = 
			COALESCE (
				OC.LSC_VALUE,
				9
			),
		OutcomeName = 
			CASE
				COALESCE (
					OC.LSC_VALUE,
					9
				)
				WHEN 1 THEN 'Achieved'
				WHEN 2 THEN 'Partially Achieved'
				WHEN 3 THEN 'Failed'
				WHEN 8 THEN 'Unknown Outcome'
				ELSE NULL
			END,
		WithdrawalReasonCode = PUS.END_REASON,
		WithdrawalReasonName = ENDR.DESCRIPTION,
		CourseStartDate = COALESCE ( UIO.FES_START_DATE, UQA.START_DATE ),
		CourseEndDate = COALESCE ( UIO.FES_END_DATE, UQA.END_DATE ),
		OriginalStartDate = PUS.ORIG_LEARN_START_DATE,
		StartDate = COALESCE ( PUS.AIM_START, PUS.START_DATE, UIO.FES_START_DATE, UQA.START_DATE ),
		PlannedEndDate = COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE ),
		ActualEndDate = 
			CASE
				WHEN CMP.LSC_VALUE = '1' THEN NULL
				ELSE PU.PROGRESS_DATE
			END,
		AchievementDate = 
			PUS.NVQ_ACHIEVE_DATE,
		IsStartThisYear = 
			CASE
				WHEN
					CASE
						WHEN 
						COALESCE ( 
							PUS.NVQ_ACHIEVE_DATE,
							CASE
								WHEN CMP.LSC_VALUE = '1' THEN NULL
								ELSE PU.PROGRESS_DATE
							END
						) > COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE ) THEN
							COALESCE ( 
								PUS.NVQ_ACHIEVE_DATE,
								CASE
									WHEN CMP.LSC_VALUE = '1' THEN NULL
									ELSE PU.PROGRESS_DATE
								END
							)
						ELSE COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE )
					END >= CAST ( CY.YrStart AS DATE ) THEN 'Y'
				ELSE 'N'
			END,
		IsPastPlannedEndDate = 
			CASE
				WHEN 
					COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE ) <= CAST ( CY.YrStart AS DATE )
					AND COALESCE (
						CMP.LSC_VALUE,
						9
					) = 1
					AND UQA.QUAL_CODE IS NOT NULL --Only include ILR quals for late finishers
				THEN 'Y'
				ELSE 'N'
			END,
		IncludeInILR = 
			CASE
				WHEN CMP.LSC_VALUE IN ( '1', '2', '3', '4', '6' ) AND UQA.QUAL_CODE IS NOT NULL THEN 'Y'
				ELSE 'N'
			END,
		FirstYear = 
			CASE
				WHEN COALESCE ( UIO.FES_START_DATE, UQA.START_DATE ) >= CAST ( CY.YrStart AS DATE ) THEN 'Y'
				ELSE 'N'
			END,
		YearNum = 
			DATEDIFF (
				YEAR,
				CASE
					WHEN MONTH( COALESCE ( UIO.FES_START_DATE, UQA.START_DATE ) ) >= 8 THEN DATEFROMPARTS(YEAR( COALESCE ( UIO.FES_START_DATE, UQA.START_DATE ) ), 8, 1)
					ELSE DATEFROMPARTS(YEAR( COALESCE ( UIO.FES_START_DATE, UQA.START_DATE ) ) - 1, 8, 1)
				END,
				CY.YrStart
			),
		OfYears = 
			DATEDIFF (
				YEAR,
				CASE
					WHEN MONTH( COALESCE ( UIO.FES_START_DATE, UQA.START_DATE ) ) >= 8 THEN DATEFROMPARTS(YEAR( COALESCE ( UIO.FES_START_DATE, UQA.START_DATE ) ), 8, 1)
					ELSE DATEFROMPARTS(YEAR( COALESCE ( UIO.FES_START_DATE, UQA.START_DATE ) ) - 1, 8, 1)
				END,
				CASE
					WHEN MONTH( COALESCE ( UIO.FES_END_DATE, UQA.END_DATE ) ) >= 8 THEN DATEFROMPARTS(YEAR( COALESCE ( UIO.FES_END_DATE, UQA.END_DATE ) ) + 1, 7, 31)
					ELSE DATEFROMPARTS(YEAR( COALESCE ( UIO.FES_END_DATE, UQA.END_DATE ) ), 7, 31)
				END
			),
		HybridEndYear = 
			CASE
				WHEN
					CASE
						WHEN 
						COALESCE ( 
							PUS.NVQ_ACHIEVE_DATE,
							CASE
								WHEN PU.PROGRESS_STATUS = 'A' THEN NULL
								ELSE PU.PROGRESS_DATE
							END
						) > COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE ) THEN
							COALESCE ( 
							PUS.NVQ_ACHIEVE_DATE,
							CASE
								WHEN PU.PROGRESS_STATUS = 'A' THEN NULL
								ELSE PU.PROGRESS_DATE
							END
						)
						ELSE COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE )
					END < '20' + LEFT ( @AcademicYear, 2 ) + '-08-01' THEN @AcademicYear
					ELSE
						CASE
							WHEN 
								MONTH (
									CASE
										WHEN 
										COALESCE ( 
											PUS.NVQ_ACHIEVE_DATE,
											CASE
												WHEN PU.PROGRESS_STATUS = 'A' THEN NULL
												ELSE PU.PROGRESS_DATE
											END
										) > COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE ) THEN
											COALESCE ( 
											PUS.NVQ_ACHIEVE_DATE,
											CASE
												WHEN PU.PROGRESS_STATUS = 'A' THEN NULL
												ELSE PU.PROGRESS_DATE
											END
										)
										ELSE COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE )
									END
								) <= 7 THEN
									RIGHT (
										CAST (
											YEAR (
												CASE
													WHEN 
													COALESCE ( 
														PUS.NVQ_ACHIEVE_DATE,
														CASE
															WHEN PU.PROGRESS_STATUS = 'A' THEN NULL
															ELSE PU.PROGRESS_DATE
														END
													) > COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE ) THEN
														COALESCE ( 
														PUS.NVQ_ACHIEVE_DATE,
														CASE
															WHEN PU.PROGRESS_STATUS = 'A' THEN NULL
															ELSE PU.PROGRESS_DATE
														END
													)
													ELSE COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE )
												END
											) - 1
										AS VARCHAR(4) )
										,
										2
									)
							ELSE
								RIGHT (
									CAST (
										YEAR (
											CASE
												WHEN 
												COALESCE ( 
													PUS.NVQ_ACHIEVE_DATE,
													CASE
														WHEN PU.PROGRESS_STATUS = 'A' THEN NULL
														ELSE PU.PROGRESS_DATE
													END
												) > COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE ) THEN
													COALESCE ( 
													PUS.NVQ_ACHIEVE_DATE,
													CASE
														WHEN PU.PROGRESS_STATUS = 'A' THEN NULL
														ELSE PU.PROGRESS_DATE
													END
												)
												ELSE COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE )
											END
										)
									AS VARCHAR(4) )
									,
									2
								)
						END
						+ '/'
						+ CASE
							WHEN 
								MONTH (
									CASE
										WHEN 
										COALESCE ( 
											PUS.NVQ_ACHIEVE_DATE,
											CASE
												WHEN PU.PROGRESS_STATUS = 'A' THEN NULL
												ELSE PU.PROGRESS_DATE
											END
										) > COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE ) THEN
											COALESCE ( 
											PUS.NVQ_ACHIEVE_DATE,
											CASE
												WHEN PU.PROGRESS_STATUS = 'A' THEN NULL
												ELSE PU.PROGRESS_DATE
											END
										)
										ELSE COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE )
									END
								) <= 7 THEN
									RIGHT (
										CAST (
											YEAR (

		AND STD.RV_DOMAIN = 'APPRENTICESHIP_TRAILBLAZER_STANDARD'
	LEFT JOIN ebs.dbo.LSC_VERIFIERS FW
		ON FW.CODE = COALESCE ( PUS.SECTOR_FRAMEWORK, UQA.SECTOR_FRAMEWORK )
		AND FW.FUNDING_YEAR = FY.FUNDING_YEAR
		AND FW.RV_DOMAIN = 'SECTOR_FRAMEWORK'
	LEFT JOIN ebs.dbo.LSC_VERIFIERS ETH
		ON ETH.CODE = COALESCE ( P.ETHNICITY, 99 )
		AND ETH.FUNDING_YEAR = FY.FUNDING_YEAR
		AND ETH.RV_DOMAIN = 'ETHNICITY'
	LEFT JOIN ebs.dbo.LSC_VERIFIERS DIF
		ON DIF.CODE = COALESCE ( P.LEARN_DIFF, 9 )
		AND DIF.FUNDING_YEAR = FY.FUNDING_YEAR
		AND DIF.RV_DOMAIN = 'LEARN_DIFF'
	LEFT JOIN ebs.dbo.VERIFIERS QT
		ON QT.LOW_VALUE = UI.FES_QUALIFICATION_TYPE
		AND QT.RV_DOMAIN = 'QUALIFICATION_TYPE'
	LEFT JOIN ebs.dbo.VERIFIERS FT
		ON FT.LOW_VALUE = UI.FES_SOURCE_FINANCE
		AND FT.RV_DOMAIN = 'SOURCE_FINANCE'
	LEFT JOIN ebs.dbo.VERIFIERS SEX
		ON SEX.LOW_VALUE = P.SEX
		AND SEX.RV_DOMAIN = 'SEX'
	LEFT JOIN ebs.dbo.VERIFIERS GEN
		ON GEN.LOW_VALUE = PAD.GENDER
		AND GEN.RV_DOMAIN = 'GENDER'
	LEFT JOIN ebs.dbo.MODES_OF_PROVISION MOA
		ON MOA.MOA_CODE = UIO.FES_MOA_CODE
	LEFT JOIN #Addresses ADM
		ON ADM.PersonCode = P.PERSON_CODE
		AND ADM.RowNum = 1
	LEFT JOIN ebs.dbo.ADDRESSES AD
		ON AD.ADDRESS_CODE = ADM.AddressCode
	LEFT JOIN #Contacts CON
		ON CON.PersonCode = P.PERSON_CODE
	LEFT JOIN #Disabilities DIS
		ON DIS.PersonCode = P.PERSON_CODE
		AND DIS.RowNum = 1
	WHERE
		(
			--Dates span current year
			(
				COALESCE ( PUS.ORIG_LEARN_START_DATE, PUS.AIM_START, PUS.START_DATE, UIO.FES_START_DATE, UQA.START_DATE ) <= CAST ( CY.YrEnd AS DATE )
				AND 
					CASE
						WHEN 
						COALESCE ( 
							PUS.NVQ_ACHIEVE_DATE,
							CASE
								WHEN CMP.LSC_VALUE = '1' THEN NULL
								ELSE PU.PROGRESS_DATE
							END
						) > COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE ) THEN
							COALESCE ( 
								PUS.NVQ_ACHIEVE_DATE,
								CASE
									WHEN CMP.LSC_VALUE = '1' THEN NULL
									ELSE PU.PROGRESS_DATE
								END
							)
						ELSE COALESCE ( PUS.AIM_END, PUS.END_DATE, UIO.FES_END_DATE, UQA.END_DATE )
					END >= CAST ( CY.YrStart AS DATE )
			)
			--Or still on programme (from previous year)
			OR (
				COALESCE ( PUS.ORIG_LEARN_START_DATE, PUS.AIM_START, PUS.START_DATE, UIO.FES_START_DATE, UQA.START_DATE ) <= CAST ( CY.YrEnd AS DATE )
				AND COALESCE (
					CMP.LSC_VALUE,
					9
				) = 1
				AND UQA.QUAL_CODE IS NOT NULL --Only include ILR quals for late finishers
			)
			--Or on a break in learning (which may be in an earlier year)
			OR (
				COALESCE ( PUS.ORIG_LEARN_START_DATE, PUS.AIM_START, PUS.START_DATE, UIO.FES_START_DATE, UQA.START_DATE ) <= CAST ( CY.YrEnd AS DATE )
				AND COALESCE (
					CMP.LSC_VALUE,
